# ==================================================
# Демонстрация всех команд эмулятора.
# Запускается с большой VFS:
#   python src/firststage.py --vfs vfs/vfs_large.zip \
#       --script scripts/demo_all_commands.txt
# ==================================================

# ----- 1. Команда pwd -----
pwd

# ----- 2. Команда ls без аргументов (текущий каталог) -----
ls

# ----- 3. Команда ls с путём -----
ls /
ls home
ls home/user

# ----- 4. Команда cd -----
cd home
pwd
cd user
pwd
cd notes
pwd

# ----- 5. Команда cat (чтение файла) -----
cat day1.md
cat day2.md
cd archive
cat old.txt

# ----- 6. Возврат в корень -----
cd /
pwd

# ----- 7. Команда vfs-init без аргументов (сброс в состояние по умолчанию) -----
vfs-init
pwd
ls

# ----- 8. Обработка ошибок -----

# 8.1. Несуществующая команда
неизвестная_команда

# 8.2. cd в несуществующий каталог
cd несуществующая_папка

# 8.3. cd с двумя аргументами (ошибка числа аргументов)
cd home user

# 8.4. cat без аргументов
cat

# 8.5. cat с двумя аргументами
cat file1 file2

# 8.6. cat несуществующего файла
cat не_существует.txt

# 8.7. Незакрытая кавычка в аргументах
cd "незакрытая

# 8.8. ls несуществующего каталога
ls /нет_такой_папки

# ----- 9. Завершение работы -----
exit
