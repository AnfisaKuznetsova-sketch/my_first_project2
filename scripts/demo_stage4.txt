# ==================================================
# Стартовый скрипт для этапа 4.
# Тестирование всех команд этапа: ls, cd, pwd, rev.
# Запуск:
#   python src/firststage.py --vfs vfs/vfs_large.zip \
#       --script scripts/demo_stage4.txt
# ==================================================

# ----- 1. pwd: показать текущий каталог -----
pwd

# ----- 2. ls: содержимое корня -----
ls

# ----- 3. cd: перейти в /home -----
cd home
pwd

# ----- 4. ls внутри /home -----
ls

# ----- 5. cd: перейти в /home/user -----
cd user
pwd

# ----- 6. ls: файлы и папки внутри /home/user -----
ls

# ----- 7. cat + rev: читаем файл и переворачиваем -----
cat readme.txt
rev readme.txt

# ----- 8. cd: перейти глубже -----
cd notes
pwd
ls

# ----- 9. rev на другом файле -----
cat day1.md
rev day1.md

# ----- 10. ls с указанием пути (без cd) -----
ls /etc
ls /var/log

# ----- 11. cd обратно в корень -----
cd /
pwd

# ----- 12. Обработка ошибок -----

# 12.1. cd в несуществующий каталог
cd несуществующая_папка

# 12.2. rev без аргументов
rev

# 12.3. rev с лишним аргументом
rev readme.txt extra

# 12.4. rev несуществующего файла
rev не_существует.txt

# 12.5. ls несуществующего каталога
ls /нет_такой_папки

# ----- 13. Завершение -----
exit
